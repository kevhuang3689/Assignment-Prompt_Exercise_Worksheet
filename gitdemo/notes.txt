Task 1:

    # Clone the remote repository to your computer
    git clone https://github.com/fortrainees2020/gitdemo.git
        * downloads repo and git history

    # Enter the project directory
    cd gitdemo

    # Check repository status
    git status

    # View configured remote repositories
    git remote -v

    # View current branch
    git branch

    # List all files, including hidden files
    ls -la

    # View recent commits
    git log --oneline


Task 2: 
    # Create a new branch
    git branch feature/my-feature

    # Switch to that branch
    git switch feature/my-feature

    # Create AND switch in one command
    git switch -c feature/my-feature

    # List local branches
    git branch

    # Switch back to main
    git switch main
